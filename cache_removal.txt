# Remove SQLite databases/caches
find . -type f \( -name "*.db" -o -name "*.db-shm" -o -name "*.db-wal" \) -delete

# Remove DiskCache / cache directories
find . -type d \( -name "cache" -o -name ".cache" -o -name "disk" -o -name "sqlite" \) -exec rm -rf {} +

# Remove cached JSON files only
find . -type f -path "*/cache/*" -name "*.json" -delete

# Remove Python cache directories
find . -type d -name "__pycache__" -exec rm -rf {} +

# Remove compiled Python files
find . -type f \( -name "*.pyc" -o -name "*.pyo" \) -delete

# Remove pytest cache
find . -type d -name ".pytest_cache" -exec rm -rf {} +

# Remove Ruff cache
find . -type d -name ".ruff_cache" -exec rm -rf {} +

# Remove mypy cache
find . -type d -name ".mypy_cache" -exec rm -rf {} +

# Remove coverage cache/files
find . -type f \( -name ".coverage" -o -name "*.coverage" \) -delete

# Remove temporary files
find . -type f \( -name "*.tmp" -o -name "*.temp" \) -delete

# Remove OS/editor temporary files
find . -type f \( -name ".DS_Store" -o -name "*~" \) -delete
